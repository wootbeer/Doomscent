// Doomscent settings (see the "Doomscent Options" menu)

// Bank the view into turns, like Descent's PF_TURNROLL.
user bool dd_bankturns = false;

// Near level flight, turn about the world's vertical axis (Doom-style) so
// the horizon stays straight.  Off: always turn about the ship's own axes,
// as in Descent.
user bool dd_levelturn = true;

// Mouse aiming with Descent-style momentum: turns ease in and out instead
// of stopping dead (the total turn is the same).
user bool dd_aimmomentum = true;

// Descent's gentle up-and-down bob of the ship.
user bool dd_shipbob = true;

// Slowly roll the ship back level when not banking.
user bool dd_autolevel = false;

// Play Descent's music instead of the game's own.
user bool dd_music = false;

// Draw the Descent cockpit (off = full-screen HUD with reticle only).
user bool dd_cockpit = true;

// Thrust multiplier.  1.0 is Descent's Pyro-GX.
server float dd_thrust = 1.0;

// Damage multiplier for Descent weapons against Doom monsters.
server float dd_damage = 1.0;

// Damage multiplier for what hits the ship.  1.0 is Doom's own damage.
server float dd_damagetaken = 1.0;

// Ick Mode: monsters wear a face (data/ickface.png), bleed green and warp
// out when they die.
server bool dd_ickmode = false;

// Open doors and press switches by flying into them.
server bool dd_bumpuse = true;
