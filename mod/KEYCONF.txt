// Doomscent controls

// The ship is the only player class.  Set here as well as in MAPINFO
// because mods like Brutal Doom pick their player class in KEYCONF, which
// GZDoom reads after MAPINFO; Doomscent's loads last, so this wins.
clearplayerclasses
addplayerclass DDShip

// Weapon slots, for the same reason (Brutal Doom sets its own here).
// 6-0 pick secondaries in DDShip.PickWeapon, so they stay empty.
weaponsection Doomscent
setslot 1 DDLaserCannon DDSuperLaserCannon
setslot 2 DDVulcanCannon DDGaussCannon
setslot 3 DDSpreadfireCannon DDHelixCannon
setslot 4 DDPlasmaCannon DDPhoenixCannon
setslot 5 DDFusionCannon DDOmegaCannon
setslot 6
setslot 7
setslot 8
setslot 9
setslot 0

addkeysection "Doomscent" DoomscentKeys

addmenukey "Bank left" +user1
addmenukey "Bank right" +user2
addmenukey "Slide up" +moveup
addmenukey "Slide down" +movedown
addmenukey "Fire flare" +user3
addmenukey "Drop bomb" +user4
addmenukey "Rear view" +dd_rearview
addmenukey "Cycle secondary" dd_nextsecondary
addmenukey "Afterburner (Descent II)" +dd_afterburner
addmenukey "Headlight (Descent II)" dd_headlight
addmenukey "Energy to shield (Descent II)" +dd_convert
addmenukey "Level ship" dd_levelship

alias dd_nextsecondary "netevent dd_nextsecondary"
alias dd_levelship "netevent dd_level"
alias dd_info "netevent dd_info"
alias +dd_rearview "netevent dd_rear 1"
alias -dd_rearview "netevent dd_rear 0"
alias +dd_afterburner "netevent dd_afterburner 1"
alias -dd_afterburner "netevent dd_afterburner 0"
alias dd_headlight "netevent dd_headlight"
alias +dd_convert "netevent dd_convert 1"
alias -dd_convert "netevent dd_convert 0"

// Bind the Descent-style layout (from the options menu).
alias dd_bindkeys "bind w +forward; bind s +back; bind a +moveleft; bind d +moveright; bind q +user1; bind e +user2; bind space +moveup; bind c +movedown; bind ctrl +movedown; bind r +use; bind f +user3; bind b +user4; bind mouse1 +attack; bind mouse2 +altattack; bind mwheelup weapprev; bind mwheeldown weapnext; bind mouse3 dd_nextsecondary; bind z dd_levelship; bind v +dd_rearview; bind shift +dd_afterburner; bind h dd_headlight; bind g +dd_convert; bind 1 \"slot 1\"; bind 2 \"slot 2\"; bind 3 \"slot 3\"; bind 4 \"slot 4\"; bind 5 \"slot 5\"; bind 6 \"slot 6\"; bind 7 \"slot 7\"; bind 8 \"slot 8\"; bind 9 \"slot 9\"; bind 0 \"slot 0\"; freelook true; lookspring false; echo Doomscent controls bound."

// Keys that are normally free in GZDoom's default layout.
defaultbind q +user1
defaultbind c +movedown
defaultbind f +user3
defaultbind b +user4
defaultbind z dd_levelship
defaultbind v +dd_rearview
defaultbind x +dd_afterburner
defaultbind h dd_headlight
defaultbind g +dd_convert
defaultbind mouse3 dd_nextsecondary
