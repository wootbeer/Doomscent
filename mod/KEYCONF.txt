// Doomscent controls

addkeysection "Doomscent" DoomscentKeys

addmenukey "Bank left" +user1
addmenukey "Bank right" +user2
addmenukey "Slide up" +moveup
addmenukey "Slide down" +movedown
addmenukey "Fire flare" +user3
addmenukey "Drop bomb" +user4
addmenukey "Cycle secondary" dd_nextsecondary

alias dd_nextsecondary "netevent dd_nextsecondary"
alias dd_levelship "netevent dd_level"

// Keys that are normally free in GZDoom's default layout.
defaultbind q +user1
defaultbind c +movedown
defaultbind f +user3
defaultbind b +user4
defaultbind z dd_levelship
defaultbind mouse3 dd_nextsecondary
